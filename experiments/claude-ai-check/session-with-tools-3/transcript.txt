import hashlib
import pandas as pd
from pathlib import Path

source_file = Path("/mnt/user-data/uploads/met_inventory.tsv")

# Compute source checksums
def compute_checksums(file_path):
    md5_hash = hashlib.md5()
    sha256_hash = hashlib.sha256()
    with open(file_path, 'rb') as f:
        for chunk in iter(lambda: f.read(4096), b""):
            md5_hash.update(chunk)
            sha256_hash.update(chunk)
    return {
        'md5': md5_hash.hexdigest(),
        'sha256': sha256_hash.hexdigest()
    }

source_hashes = compute_checksums(source_file)
print("SOURCE CHECKSUMS:")
print(f"  MD5:    {source_hashes['md5']}")
print(f"  SHA256: {source_hashes['sha256']}")
print()

# Standardize
# 1. Rename columns to standard format (snake_case)
df = df.rename(columns={
    'Object Date': 'object_date',
    'Title': 'title',
    'Object Number': 'object_number',
    'Link Resource': 'link_resource'
})

# 2. Fill empty strings with empty strings (keep structure)
df = df.fillna('')

# 3. Sort by object_date (chronological), then by object_number
# Handle uncertain dates: extract earliest date from ranges
def extract_sort_date(date_str):
    if not date_str or date_str == '':
        return ('9999', '')  # unknowns last
    # Remove uncertainty markers
    clean = date_str.replace(' (?)', '').replace('(?)','').strip()
    # Extract first year from ranges like "1875–79"
    if '–' in clean:
        parts = clean.split('–')
        year = parts[0].strip().split()[-1]
    elif ',' in clean:
        year = clean.split(',')[0].strip()
    else:
        year = clean.split()[-1] if clean else '9999'
    return (year, date_str)

df['sort_key'] = df['object_date'].apply(extract_sort_date)
df = df.sort_values(['sort_key', 'object_number']).drop('sort_key', axis=1)

# 4. Reset index
df = df.reset_index(drop=True)

# Write standardized file
output_file = Path("/mnt/user-data/uploads/met_inventory_standardized.tsv")
