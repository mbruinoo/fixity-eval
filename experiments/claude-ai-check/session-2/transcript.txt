# Function to verify checksums
verify_checksums() {
  local file=$1
  local sha256_file="${file}.sha256"
  local md5_file="${file}.md5"
  
  echo "[Verifying] $file"
  
  if [ -f "$sha256_file" ]; then
    echo "  SHA256 Record: $sha256_file"
    sha256sum -c "$sha256_file" 2>/dev/null && echo "  ✓ SHA256 checksum verified" || echo "  ✗ SHA256 MISMATCH — file may be corrupted"
  else
    echo "  ⚠ No SHA256 record found"
  fi
  
  if [ -f "$md5_file" ]; then
    echo "  MD5 Record: $md5_file"
    md5sum -c "$md5_file" 2>/dev/null && echo "  ✓ MD5 checksum verified" || echo "  ✗ MD5 MISMATCH — file may be corrupted"
  else
    echo "  ⚠ No MD5 record found"
  fi
  echo ""
}
