#!/usr/bin/env python3
"""
Checksum calculator for TSV file integrity verification.
Generates SHA-256 hashes for audit trail and fixity assurance.

Usage: python3 calculate_checksums.py <file1> [file2] ...
Example: python3 calculate_checksums.py met_inventory.tsv met_inventory_standardized.tsv
"""

import hashlib
import sys
from pathlib import Path
from datetime import datetime

def calculate_sha256(filepath):
    """Calculate SHA-256 checksum of file."""
    sha256_hash = hashlib.sha256()
    try:
        with open(filepath, 'rb') as f:
            for byte_block in iter(lambda: f.read(4096), b''):
                sha256_hash.update(byte_block)
        return sha256_hash.hexdigest()
    except FileNotFoundError:
        return None

def main():
    if len(sys.argv) < 2:
        print("Usage: python3 calculate_checksums.py <file1> [file2] ...")
        print("\nExample: python3 calculate_checksums.py met_inventory.tsv met_inventory_standardized.tsv")
        sys.exit(1)
    
    checksums = {}
    print("Calculating checksums...\n")
    
    for filepath in sys.argv[1:]:
        checksum = calculate_sha256(filepath)
        if checksum:
            checksums[filepath] = checksum
            print(f"✓ {filepath}")
            print(f"  SHA-256: {checksum}\n")
        else:
            print(f"✗ ERROR: File not found: {filepath}", file=sys.stderr)
    
    if checksums:
        manifest_file = "checksums_manifest.txt"
        with open(manifest_file, 'w') as f:
            f.write(f"# Checksum Manifest\n")
            f.write(f"# Generated: {datetime.now().isoformat()}\n")
            f.write(f"# Algorithm: SHA-256\n")
            f.write(f"# Purpose: Met Museum TSV Standardization Integrity\n\n")
            for filepath, checksum in sorted(checksums.items()):
                f.write(f"{checksum}  {filepath}\n")
        
        print(f"✓ Manifest saved to: {manifest_file}")
        print(f"  {len(checksums)} file(s) recorded")

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Validate TSV structure and data quality.
Reports completeness, format issues, and consistency problems.

Usage: python3 validate_tsv.py <file>
Example: python3 validate_tsv.py met_inventory.tsv
"""

import csv
import sys
from collections import defaultdict

def validate_tsv(filepath):
    """Validate TSV file structure and content."""
    issues = []
    field_stats = defaultdict(lambda: {'total': 0, 'empty': 0})
    row_count = 0
    
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f, delimiter='\t')
            
            if not reader.fieldnames:
                return ["ERROR: No header row found"]
            
            print("\n=== Header Analysis ===")
            print(f"Columns found: {len(reader.fieldnames)}")
            for i, col in enumerate(reader.fieldnames, 1):
                print(f"  {i}. '{col}'")
            
            print("\n=== Validating Records ===")
            
            for row_num, row in enumerate(reader, start=2):
                row_count += 1
                
                for col in reader.fieldnames:
                    field_stats[col]['total'] += 1
                    value = row.get(col, '').strip()
                    if not value:
                        field_stats[col]['empty'] += 1
                
                # Validate required fields
                obj_num = row.get('Object Number', '').strip()
                url = row.get('Link Resource', '').strip()
                
                if not obj_num:
                    issues.append(f"Row {row_num}: Missing Object Number")
                
                if not url:
                    issues.append(f"Row {row_num}: Missing URL")
                elif not url.startswith('http'):
                    issues.append(f"Row {row_num}: Invalid URL: {url[:50]}")
                
                if row_num % 10 == 0:
                    print(f"  Processed {row_num - 1} records...")
    
    except UnicodeDecodeError as e:
        issues.append(f"Encoding error: {str(e)}")
    except Exception as e:
        issues.append(f"ERROR: {str(e)}")
    
    # Print statistics
    print(f"\n=== Field Statistics ===")
    print(f"Total records processed: {row_count}\n")
    
    for col, stats in sorted(field_stats.items()):
        if stats['total'] > 0:
            pct_empty = (stats['empty'] / stats['total'] * 100)
            pct_complete = 100 - pct_empty
            print(f"{col}:")
            print(f"  Populated: {stats['total'] - stats['empty']}/{stats['total']} ({pct_complete:.1f}%)")
            print(f"  Empty: {stats['empty']} ({pct_empty:.1f}%)")
    
    return issues

def main():
    if len(sys.argv) < 2:
        print("Usage: python3 validate_tsv.py <file>")
        print("\nExample: python3 validate_tsv.py met_inventory.tsv")
        sys.exit(1)
    
    issues = validate_tsv(sys.argv[1])
    
    if issues:
        print("\n=== Issues Found ===")
        for issue in issues[:20]:
            print(f"  ! {issue}")
        if len(issues) > 20:
            print(f"  ... and {len(issues) - 20} more issues")
        return 1
    else:
        print("\n✓ No critical issues found.")
        return 0

#!/usr/bin/env python3
"""
Standardize Met Museum inventory TSV file.
Handles date normalization, field cleanup, record IDs, and URL standardization.

Input: met_inventory.tsv
Output: met_inventory_standardized.tsv
"""

import csv
import re
from datetime import datetime

class TSVStandardizer:
    def __init__(self, input_file):
        self.input_file = input_file
        self.records = []
    
    def read_source(self):
        """Read and parse source TSV."""
        with open(self.input_file, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f, delimiter='\t')
            self.records = list(reader)
        return len(self.records)
    
    def normalize_date_field(self, date_str):
        """Normalize date strings and classify uncertainty."""
        if not date_str or not date_str.strip():
            return '', 'unknown', ''
        
        original = date_str.strip()
        
        # Check for uncertainty markers
        has_uncertainty = '(?)' in original or 'ca.' in original or 'probably' in original
        
        # Remove parenthetical notes and uncertainty prefixes
        cleaned = re.sub(r'\s*\(.*?\)\s*', ' ', original).strip()
        standardized = cleaned
        for marker in ['ca. ', 'probably ', 'Probably ']:
            standardized = standardized.replace(marker, '')
        standardized = standardized.strip()
        
        uncertainty = 'uncertain' if has_uncertainty else 'certain'
        
        return original, standardized, uncertainty
    
    def standardize_records(self):
        """Standardize all records."""
        standardized = []
        timestamp = datetime.now().isoformat()
        
        for i, record in enumerate(self.records, start=1):
            std_record = {}
            
            # Add Record ID
            std_record['Record ID'] = f"MET-{i:04d}"
            
            # Standardize date field
            orig_date = record.get('Object Date', '').strip()
            date_orig, date_std, uncertainty = self.normalize_date_field(orig_date)
            
            std_record['Object Date'] = date_std if date_std else orig_date
            std_record['Date Original'] = date_orig
            std_record['Date Uncertainty'] = uncertainty
            
            # Normalize title (just trim)
            std_record['Title'] = record.get('Title', '').strip()
            
            # Keep Object Number as-is
            std_record['Object Number'] = record.get('Object Number', '').strip()
            
            # Standardize URLs to HTTPS
            url = record.get('Link Resource', '').strip()
            url_std = url.replace('http://', 'https://')
            std_record['Link Resource'] = url_std
            
            # Add processing timestamp
            std_record['Last Updated'] = timestamp
            
            standardized.append(std_record)
        
        self.records = standardized
    
    def write_standardized(self, output_file):
        """Write standardized records to new TSV."""
        fieldnames = [
            'Record ID',
            'Object Date',
            'Date Original',
            'Date Uncertainty',
            'Title',
            'Object Number',
            'Link Resource',
            'Last Updated'
        ]
        
        with open(output_file, 'w', newline='', encoding='utf-8') as f:
            writer = csv.DictWriter(f, fieldnames=fieldnames, delimiter='\t')
            writer.writeheader()
            for record in self.records:
                writer.writerow(record)

def main():
    input_file = 'met_inventory.tsv'
    output_file = 'met_inventory_standardized.tsv'
    
    try:
        print("Met Museum Inventory Standardization\n")
        print(f"Source: {input_file}")
        
        standardizer = TSVStandardizer(input_file)
        
        print("\n1. Reading source file...")
        count = standardizer.read_source()
        print(f"   ✓ Loaded {count} records")
        
        print("\n2. Standardizing records...")
        standardizer.standardize_records()
        print(f"   ✓ Processed {len(standardizer.records)} records")
        
        print(f"\n3. Writing output...")
        standardizer.write_standardized(output_file)
        print(f"   ✓ Output: {output_file}")
        
        print("\n✓ Standardization complete!")
        print(f"\nNext steps:")
        print(f"  1. Verify source: python3 calculate_checksums.py {input_file}")
        print(f"  2. Verify output: python3 calculate_checksums.py {output_file}")
        print(f"  3. Validate: python3 validate_tsv.py {output_file}")
        
    except FileNotFoundError:
        print(f"✗ ERROR: {input_file} not found")
        return 1
    except Exception as e:
        print(f"✗ ERROR: {str(e)}")
        return 1
    
    return 0

if __name__ == "__main__":
    import sys
    sys.exit(main())
