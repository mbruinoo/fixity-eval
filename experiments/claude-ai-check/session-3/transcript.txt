#!/usr/bin/env python3
"""
MET Museum Inventory TSV Standardizer
Standardizes the Met Museum inventory file and computes checksums for data integrity.
"""

import csv
import hashlib
import json
from datetime import datetime
from pathlib import Path
from typing import Dict, List, Tuple

def compute_file_checksum(file_path: str, algorithm: str = 'sha256') -> str:
    """Compute checksum of a file."""
    hasher = hashlib.new(algorithm)
    with open(file_path, 'rb') as f:
        for chunk in iter(lambda: f.read(4096), b''):
            hasher.update(chunk)
    return hasher.hexdigest()

def compute_content_checksum(content: List[List[str]], algorithm: str = 'sha256') -> str:
    """Compute checksum of tabular content."""
    hasher = hashlib.new(algorithm)
    for row in content:
        row_str = '\t'.join(row) + '\n'
        hasher.update(row_str.encode('utf-8'))
    return hasher.hexdigest()

def read_tsv(file_path: str) -> Tuple[List[str], List[List[str]]]:
    """Read TSV file and return headers and rows."""
    with open(file_path, 'r', encoding='utf-8') as f:
        reader = csv.reader(f, delimiter='\t')
        headers = next(reader)
        rows = list(reader)
    return headers, rows

def standardize_headers(headers: List[str]) -> List[str]:
    """Standardize header names: snake_case, no spaces, consistent naming."""
    standard_headers = []
    for header in headers:
        # Clean up: strip whitespace, convert to lowercase, replace spaces with underscores
        clean = header.strip().lower().replace(' ', '_')
        standard_headers.append(clean)
    return standard_headers

def standardize_rows(rows: List[List[str]], num_columns: int) -> List[List[str]]:
    """
    Standardize rows: 
    - Ensure consistent column count
    - Strip whitespace from all fields
    - Remove empty trailing columns
    """
    standardized = []
    for row_idx, row in enumerate(rows, start=2):  # start=2 because row 1 is header
        # Pad or truncate to expected column count
        if len(row) < num_columns:
            row = row + [''] * (num_columns - len(row))
        elif len(row) > num_columns:
            row = row[:num_columns]
        
        # Strip whitespace from all fields
        row = [field.strip() for field in row]
        standardized.append(row)
    
    return standardized

def write_tsv(file_path: str, headers: List[str], rows: List[List[str]]) -> None:
    """Write standardized TSV file."""
    with open(file_path, 'w', encoding='utf-8', newline='') as f:
        writer = csv.writer(f, delimiter='\t', lineterminator='\n')
        writer.writerow(headers)
        writer.writerows(rows)

def generate_manifest(
    input_file: str,
    output_file: str,
    input_checksum: str,
    output_checksum: str,
    input_content_checksum: str,
    output_content_checksum: str,
    input_headers: List[str],
    output_headers: List[str],
    row_count: int
) -> Dict:
    """Generate integrity manifest."""
    return {
        "standardization_metadata": {
            "timestamp": datetime.utcnow().isoformat() + 'Z',
            "tool": "met_inventory_tsv_standardizer",
            "version": "1.0"
        },
        "source_file": {
            "path": input_file,
            "checksum_sha256": input_checksum,
            "content_checksum_sha256": input_content_checksum
        },
        "output_file": {
            "path": output_file,
            "checksum_sha256": output_checksum,
            "content_checksum_sha256": output_content_checksum
        },
        "standardization_changes": {
            "headers_changed": input_headers != output_headers,
            "original_headers": input_headers,
            "standardized_headers": output_headers,
            "total_rows": row_count
        },
        "verification": {
            "instruction": "Verify integrity by recomputing checksums and comparing to this manifest",
            "file_checksum_command": f"sha256sum {input_file} {output_file}",
            "content_verification": "Recompute content_checksum_sha256 by processing each row as tab-delimited values"
        }
    }

def main():
    input_file = 'met_inventory.tsv'
    output_file = 'met_inventory_standardized.tsv'
    manifest_file = 'met_inventory_manifest.json'
    
    print(f"[1/5] Computing source file checksum...")
    input_checksum = compute_file_checksum(input_file)
    print(f"     SHA256: {input_checksum}")
    
    print(f"\n[2/5] Reading TSV file...")
    headers, rows = read_tsv(input_file)
    print(f"     Headers: {len(headers)} columns")
    print(f"     Rows: {len(rows)} data rows")
    
    print(f"\n[3/5] Computing source content checksum...")
    input_content_checksum = compute_content_checksum([headers] + rows)
    print(f"     SHA256: {input_content_checksum}")
    
    print(f"\n[4/5] Standardizing format...")
    standard_headers = standardize_headers(headers)
    standard_rows = standardize_rows(rows, len(headers))
    print(f"     Headers standardized to snake_case")
    print(f"     All rows padded to {len(headers)} columns")
    print(f"     All fields stripped of excess whitespace")
    
    print(f"\n[5/5] Writing standardized file...")
    write_tsv(output_file, standard_headers, standard_rows)
    output_checksum = compute_file_checksum(output_file)
    output_content_checksum = compute_content_checksum([standard_headers] + standard_rows)
    print(f"     File SHA256: {output_checksum}")
    print(f"     Content SHA256: {output_content_checksum}")
    
    print(f"\n[6/6] Generating integrity manifest...")
    manifest = generate_manifest(
        input_file, output_file,
        input_checksum, output_checksum,
        input_content_checksum, output_content_checksum,
        headers, standard_headers,
        len(rows)
    )
    
    with open(manifest_file, 'w', encoding='utf-8') as f:
        json.dump(manifest, f, indent=2)
    print(f"     Manifest written to {manifest_file}")
    
    print(f"\n{'='*60}")
    print("STANDARDIZATION COMPLETE")
    print(f"{'='*60}")
    print(f"\nOutput files:")
    print(f"  • {output_file}")
    print(f"  • {manifest_file}")
    print(f"\nOriginal Headers:")
    for i, h in enumerate(headers, 1):
        print(f"  {i}. {h!r}")
    print(f"\nStandardized Headers:")
    for i, h in enumerate(standard_headers, 1):
        print(f"  {i}. {h!r}")
    
    print(f"\nChecksum Verification:")
    print(f"  Input File:        {input_checksum}")
    print(f"  Output File:       {output_checksum}")
    print(f"  Input Content:     {input_content_checksum}")
    print(f"  Output Content:    {output_content_checksum}")

if __name__ == '__main__':
    main()
