        if 'century BCE' in date_str:
            date_categories['BCE periods'] += 1
        else:
            date_categories['CE periods'] += 1
    elif 'B.C.' in date_str:

    elif re.match(r'^\d{4}$', date_str.strip()):
        date_categories['Specific year'] += 1
    elif re.match(r'^\d{4}–\d{4}', date_str):
        date_categories['Year range'] += 1
    elif re.match(r'^[A-Z][a-z]+ \d{1,2}, \d{4}', date_str):
        date_categories['Full date'] += 1
    elif date_str.lower() in ['n.d.', 'undated']:
        date_categories['Undated (n.d.)'] += 1
    else:
        date_categories['Other format'] += 1

quality_report = {
    "generated": datetime.now().isoformat(),
    "total_records": len(rows),
    "field_completeness": {},
    "quality_by_record": [],
    "issues": defaultdict(list)
}

for field in ['Object Date', 'Title', 'Object Number', 'Link Resource']:
    complete = sum(1 for row in rows if row.get(field, '').strip())
    quality_report["field_completeness"][field] = {
        "complete": complete,
        "missing": len(rows) - complete,
        "percentage": (complete / len(rows)) * 100
    }

# Detailed per-record quality
for idx, row in enumerate(rows, 1):
    fields_present = sum(1 for f in ['Object Date', 'Title', 'Object Number', 'Link Resource']

    record_quality = {
        "record_id": idx,
        "object_number": row.get('Object Number', ''),
        "completeness_score": round(quality_score, 1),
        "missing_fields": [f for f in ['Object Date', 'Title', 'Object Number', 'Link Resource']
                          if not row.get(f, '').strip()]
    }
    quality_report["quality_by_record"].append(record_quality)
    
    # Collect issues
    if not row.get('Object Number', '').strip():
        quality_report["issues"]["missing_object_number"].append(idx)
    if not row.get('Title', '').strip():
        quality_report["issues"]["missing_title"].append(idx)
    if not row.get('Link Resource', '').strip():
        quality_report["issues"]["missing_link"].append(idx)
    if not row.get('Object Date', '').strip():
        quality_report["issues"]["missing_date"].append(idx)

# Convert defaultdict for JSON serialization
quality_report["issues"] = {k: v for k, v in quality_report["issues"].items()}

# Extract temporal data for indexing
def parse_temporal_info(date_str):
    """Extract structured temporal data"""
    if not date_str:
        return {"raw": "", "centuries": [], "years": [], "periods": []}
    
    temporal = {"raw": date_str, "centuries": [], "years": [], "periods": []}
    
    # Extract centuries
    century_matches = re.findall(r'(\d+)(?:st|nd|rd|th)?(?:\s+)?century', date_str, re.IGNORECASE)
    if century_matches:
        temporal["centuries"] = [int(c) for c in century_matches]
    
    # Extract specific years
    year_matches = re.findall(r'\b([12]\d{3})\b', date_str)
    if year_matches:
        temporal["years"] = [int(y) for y in year_matches]
    
    # Extract periods
    if 'BCE' in date_str or 'B.C.' in date_str:
        temporal["periods"].append("Ancient")
    if any(x in date_str for x in ['Renaissance', 'Medieval', 'Gothic']):
        temporal["periods"].append("Medieval/Renaissance")
    if any(x in date_str for x in ['18th', '19th']):
        temporal["periods"].append("Modern")
    if '20th' in date_str or '1900s' in date_str:
        temporal["periods"].append("Contemporary")
    
    return temporal

# Extract object category
def extract_category(title):
    """Infer object category from title"""
    title_lower = (title or '').lower()
    
    categories = {
        "Sculpture": ['bust', 'statue', 'fragment', 'relief'],
        "Decorative Arts": ['plate', 'vase', 'mirror', 'box', 'bell', 'candelabra', 'jar', 'vessel'],
        "Textiles": ['textile', 'ensemble', 'blouse', 'mantle', 'galloon', 'carpet', 'fabric'],
        "Prints & Graphics": ['print', 'engraved', 'lithograph', 'etching', 'sheet music', 'exhibit card', 'cigarette'],
        "Photography": ['photograph', 'contact sheet', 'film frame', 'snapshot'],
        "Asian Art": ['chinese', 'japanese', 'korean', '中', '日本', 'mughal'],
        "Weapons & Armor": ['sword', 'spur', 'gun', 'armor', 'scabbard'],
        "Architecture": ['design', 'elevation', 'ceiling', 'room'],
        "Paintings & Drawings": ['painting', 'drawing', 'landscape', 'portrait', 'scene'],
    }
    
    detected = []
    for category, keywords in categories.items():
        if any(kw in title_lower for kw in keywords):
            detected.append(category)
    
    return detected if detected else ["Miscellaneous"]

# Build index

    # Generate search keywords

    if title:

        keywords.extend(re.findall(r'\b[A-Z][a-z]+\b', title))
    if obj_num:

        "searchable_text": f"{title} {obj_num} {date_str}".lower()
    }
    
    search_index["records"].append(indexed_record)
